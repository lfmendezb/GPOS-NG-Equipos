/// <summary>Hallazgos que necesitan la base de sistema temporal (GPOS_TEST_SIS_*) de <see cref="MantenimientoAnilloTests"/>.</summary>
public sealed partial class MantenimientoAnilloTests
{
    /// <summary>SF-01 (a): <c>anillo --cambiar-recuperacion</c> ancla el .cer que esté configurado, sin huella dictada por un segundo canal (SA-01 no lo cubre).</summary>
    [Fact]
    public async Task SF01a_cambiar_recuperacion_ancla_un_cer_sustituido_sin_huella_dictada()
    {
        using (var p = _i.Proceso())
            await Verbos(p).AnclarAsync("ACTA-IMPL-1", "implantación de prueba", _i.HuellaRecuperacion);
        // El .cer «nuevo» que llegó por el mismo canal que el primero, sustituido en tránsito: nadie compara su huella con una dictada
        using var ajeno = CertificadosPrueba.Crear("GPOS prueba recuperacion AJENA SF-01");
        _i.Origen.RecuperacionActual = CertificadosPrueba.SoloPublica(ajeno);
        using (var p = _i.Proceso())
            await Verbos(p).CambiarRecuperacionAsync(null, "ACTA-MANT-1", "cambio de custodia");
        using (var p = _i.Proceso())
        {
            var r = await Verbos(p).ComprobarAsync();
            Assert.True(r.Cumple);
            Assert.Equal(CertificadosAnillo.Huella(ajeno), r.HuellaAnclada);   // el ancla consagra la recuperación ajena
        }
        // Y la llave predeterminada nueva queda envuelta también para la recuperación ajena
        Assert.Contains(ArchivoAnillo.LlavesConformes(_i.Carpeta), l => l.Destinatarios!.Recuperacion == CertificadosAnillo.Huella(ajeno));
    }

    /// <summary>SF-01 (b): <c>anillo --migrar</c> sin ancla previa ancla la recuperación configurada sin huella dictada (lo que B señaló en su aviso).</summary>
    [Fact]
    public async Task SF01b_migrar_desde_DPAPI_ancla_sin_huella_dictada()
    {
        if (!OperatingSystem.IsWindows()) return;
        await InstalacionDpapiAsync();
        ProtegerCarpeta();
        using var ajeno = CertificadosPrueba.Crear("GPOS prueba recuperacion AJENA SF-01b");
        _i.Origen.RecuperacionActual = CertificadosPrueba.SoloPublica(ajeno);
        using (var p = _i.Proceso())
            await Verbos(p).MigrarDesdeDpapiAsync(null, "ACTA-MIG-1", "corte de la entrega 1");
        using (var p = _i.Proceso())
            Assert.Equal(CertificadosAnillo.Huella(ajeno), (await Verbos(p).ComprobarAsync()).HuellaAnclada);
    }

    /// <summary>
    /// SF-02: <c>sistema</c> sin <c>--cuentas-iniciales</c> ni <c>--confirmar-sellado</c> escribe <c>Licencia.InstalacionId</c> con el anillo del proceso,
    /// pero G-1 no lo cubre. Con otro anillo (el DPAPI del técnico), la fila queda escrita y el anillo del servicio ya no la verifica ni la reemplaza.
    /// </summary>
    [Fact]
    public async Task SF02_sistema_sin_banderas_no_pasa_por_G1_y_escribe_InstalacionId_con_el_anillo_del_tecnico()
    {
        Assert.False(GuardasAnillo.UsaElAnillo("sistema", ["login-api"]));
        Assert.False(GuardasAnillo.UsaElAnillo("sistema", []));
        var tecnico = new ServiceCollection();
        ProteccionDatosInstalacion.Configurar(tecnico, AnilloPrueba.Config(_i.Carpeta + "-tecnico", null, demo: true));
        await using (var p = tecnico.BuildServiceProvider())
        await using (var cn = new SqlConnection(_conexion))
        {
            await cn.OpenAsync();
            Assert.Contains("generado", await ComandoSistema.AsegurarInstalacionAsync(cn, p.GetRequiredService<IDataProtectionProvider>()));
        }
        // El servicio (anillo con certificado, anclado) ya no la verifica y no la reemplaza: identidad de la instalación rota
        using (var p = _i.Proceso())
        {
            await Verbos(p).AnclarAsync("ACTA-IMPL-1", "implantación de prueba", _i.HuellaRecuperacion);
            await Verbos(p).ComprobarAsync();
            await using var cn = new SqlConnection(_conexion);
            await cn.OpenAsync();
            var ex = await Assert.ThrowsAsync<GPOS.Core.Infraestructura.ReglaNegocioException>(() =>
                ComandoSistema.AsegurarInstalacionAsync(cn, p.GetRequiredService<IDataProtectionProvider>()));
            Assert.Contains("no verifica", ex.Message);
        }
        if (Directory.Exists(_i.Carpeta + "-tecnico")) Directory.Delete(_i.Carpeta + "-tecnico", recursive: true);
    }

    /// <summary>
    /// SF-03: orden de 3.3 (PR-D2, paso 4): <c>sistema --login-api</c> va ANTES de <c>anillo --anclar</c> (que necesita la base que crea <c>sistema</c>).
    /// Con el anillo con certificado ya preparado, sin llaves y sin ancla, <c>sistema</c> llama siempre a <c>AsegurarInstalacionAsync</c> (G-2 no corre sin
    /// <c>--cuentas-iniciales</c>) y la primera protección falla con <c>AnilloSinAncla</c>: el verbo sale con error a mitad (base y S1 hechos, sin
    /// <c>InstalacionId</c> ni marca). Después de <c>--anclar</c> (que crea la primera llave al proteger la fila del ancla), sí escribe.
    /// </summary>
    [Fact]
    public async Task SF03_sistema_login_api_antes_de_anclar_falla_en_una_instalacion_nueva_con_el_anillo_con_certificado()
    {
        Assert.False(Directory.Exists(_i.Carpeta) && _i.ArchivosLlave().Length > 0);
        using (var p = _i.Proceso())
        await using (var cn = new SqlConnection(_conexion))
        {
            await cn.OpenAsync();
            var ex = await Assert.ThrowsAnyAsync<Exception>(() => ComandoSistema.AsegurarInstalacionAsync(cn, p.GetRequiredService<IDataProtectionProvider>()));
            Assert.Contains("AnilloSinAncla", ex.ToString());
        }
        using (var p = _i.Proceso())
            await Verbos(p).AnclarAsync("ACTA-IMPL-1", "implantación de prueba", _i.HuellaRecuperacion);
        Assert.NotEmpty(_i.ArchivosLlave());   // --anclar creó la primera llave al proteger la fila del ancla
        using (var p = _i.Proceso())
        await using (var cn = new SqlConnection(_conexion))
        {
            await cn.OpenAsync();
            Assert.Contains("generado", await ComandoSistema.AsegurarInstalacionAsync(cn, p.GetRequiredService<IDataProtectionProvider>()));
        }
    }
}

/// <summary>Hallazgos que no necesitan SQL Server ni la máquina.</summary>
[Trait("Esquema", "Neutral")]
[Trait("Categoria", "Anillo")]
[SupportedOSPlatform("windows")]
public sealed class RevisionAdversarialF1Tests : IDisposable
{
    private readonly string _base = Directory.CreateTempSubdirectory("gpos-c-f1-").FullName;
    private string Raiz => Path.Combine(_base, "GPOS NG");
    private string RutaArchivo => Path.Combine(Raiz, "config", "GPOS.Api.anillo.json");
    private string CarpetaLlaves => Path.Combine(Raiz, "Api", "anillo");

    public void Dispose() => Directory.Delete(_base, recursive: true);

    /// <summary>
    /// SF-04: sin el archivo, sin llaves en Api\anillo, con <c>DpapiDemo</c> y con la base que no responde al arrancar, la regla del ancla de SA-03 no
    /// se aplica: solo queda un aviso y la API arranca con DPAPI (lo que B señaló). Con la base caída la API no sirve, pero no es falla cerrada.
    /// </summary>
    [Fact]
    public async Task SF04_si_la_fila_del_ancla_no_se_puede_leer_la_API_arranca_con_DPAPI()
    {
        var config = new ConfigurationBuilder().AddInMemoryCollection(new Dictionary<string, string?> { [ProteccionDatosInstalacion.ClaveDpapiDemo] = "true" })
            .AgregarArchivoAnillo("GPOS.Api", RutaArchivo, (_, _) => null).Build();
        var servicios = new ServiceCollection();
        servicios.AddSingleton<IConfiguration>(config);
        servicios.AddDbContext<SistemaDbContext>(o => o.UseSqlServer("Server=tcp:127.0.0.1,1;Database=GPOS_SYSDATA;Connect Timeout=2;TrustServerCertificate=true"));
        ProteccionDatosInstalacion.Configurar(servicios, config);
        await using var p = servicios.BuildServiceProvider();
        var r = await ComprobacionAnillo.EjecutarAlArrancarAsync(p);   // no lanza
        Assert.False(r.Activo);
    }

    /// <summary>
    /// SF-05: una instalación con el anillo con certificado configurado por claves sueltas (el modo anterior a F1, que la regla 1 dice que «sigue como
    /// hoy») y con sus llaves en Api\anillo deja de arrancar en cuanto se actualiza a F1, porque no tiene el archivo (SA-03).
    /// </summary>
    [Fact]
    public void SF05_el_anillo_configurado_por_claves_sueltas_con_llaves_en_Api_anillo_deja_de_arrancar_con_F1()
    {
        using var servidor = CertificadosPrueba.Crear("GPOS prueba servidor SF-05");
        using var recuperacion = CertificadosPrueba.Crear("GPOS prueba recuperacion SF-05");
        var huella = CertificadosAnillo.Huella(servidor);
        // Antes de F1: anillo con certificado por claves sueltas, en la carpeta de ADR-77
        var antes = new ServiceCollection();
        antes.AddSingleton<IOrigenCertificadosAnillo>(new OrigenPrueba(servidor, CertificadosPrueba.SoloPublica(recuperacion)));
        ProteccionDatosInstalacion.Configurar(antes, AnilloPrueba.Config(CarpetaLlaves, huella));
        using (var p = antes.BuildServiceProvider())
        {
            p.GetRequiredService<EstadoAnclaAnillo>().Fijar(SituacionAncla.Anclada, CertificadosAnillo.Huella(recuperacion));
            p.GetRequiredService<IKeyManager>().CreateNewKey(DateTimeOffset.UtcNow, DateTimeOffset.UtcNow.AddDays(90));
        }
        Assert.Single(ArchivoAnillo.LlavesConformes(CarpetaLlaves));
        // Después de F1: las mismas claves sueltas, más la fuente del archivo (ausente), como hace Program.cs
        var config = new ConfigurationBuilder()
            .AddInMemoryCollection(new Dictionary<string, string?>
            {
                [ProteccionDatosInstalacion.ClaveCarpeta] = CarpetaLlaves, [OrigenCertificadosAlmacen.ClaveServidor] = huella,
            })
            .AgregarArchivoAnillo("GPOS.Api", RutaArchivo, (_, _) => null).Build();
        var ex = Assert.Throws<ArchivoAnilloNoConformeException>(() => ProteccionDatosInstalacion.Configurar(new ServiceCollection(), config));
        Assert.Contains("anillo --preparar", ex.Message);
    }

    /// <summary>
    /// Control (no hallazgo): en el anfitrión real de ASP.NET, el <see cref="IConfiguration"/> de los servicios es un <see cref="IConfigurationRoot"/> con
    /// el proveedor del archivo, así que la regla del ancla de SA-03 en <c>EjecutarAlArrancarAsync</c> sí se alcanza.
    /// </summary>
    [Fact]
    public void Control_el_anfitrion_real_expone_el_proveedor_del_archivo_a_la_comprobacion_del_arranque()
    {
        var b = WebApplication.CreateBuilder();
        b.Configuration.AgregarArchivoAnillo("GPOS.Api", RutaArchivo, (_, _) => null);
        using var app = b.Build();
        var p = ArchivoAnillo.Proveedor(app.Services.GetRequiredService<IConfiguration>());
        Assert.NotNull(p);
        Assert.False(p!.Presente);
    }

    /// <summary>
    /// Control: el conflicto con otra fuente con el MISMO valor se acepta (D-AN-05 resuelto como «otro valor»), y con otro valor se rechaza sin
    /// volcar el valor en el mensaje (SA-11 cerrado para el 4704).
    /// </summary>
    [Fact]
    public void Control_conflicto_de_fuentes_mismo_valor_admitido_y_otro_valor_sin_volcar_el_valor()
    {
        Directory.CreateDirectory(Path.GetDirectoryName(RutaArchivo)!);
        var h = new string('a', 64);
        var contenido = new ContenidoArchivoAnillo("GPOS.Api", CarpetaLlaves, h, [], Path.Combine(Raiz, "config", "GPOS.Api.recuperacion.cer"), @"NT SERVICE\GPOS.Api");
        File.WriteAllBytes(RutaArchivo, contenido.Serializar());
        IConfigurationRoot Con(string valor) => new ConfigurationBuilder()
            .AddInMemoryCollection(new Dictionary<string, string?> { [OrigenCertificadosAlmacen.ClaveServidor] = valor })
            .AgregarArchivoAnillo("GPOS.Api", RutaArchivo, (_, _) => null).Build();
        Assert.NotNull(ArchivoAnillo.Validar(Con(h.ToUpperInvariant())));   // mismo valor (sin distinguir mayúsculas): admitido
        const string Centinela = "bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb";
        var ex = Assert.Throws<ArchivoAnilloNoConformeException>(() => ArchivoAnillo.Validar(Con(Centinela)));
        Assert.DoesNotContain(Centinela, ex.Message);
        Assert.DoesNotContain(h, ex.Message);
    }

    /// <summary>
    /// SF-06: el archivo admite <c>carpeta</c> y <c>certificadoRecuperacion</c> en cualquier ruta absoluta (incluida una UNC) y huellas repetidas o
    /// iguales a la vigente en <c>certificadosServidorAnteriores</c> (D-AN-13 y D-AN-14 sin cerrar). Solo un administrador escribe el archivo.
    /// </summary>
    [Fact]
    public void SF06_el_esquema_admite_rutas_UNC_y_huellas_anteriores_repetidas()
    {
        var h = new string('c', 64);
        var c = new ContenidoArchivoAnillo("GPOS.Api", @"\\servidor\recurso\anillo", h, [h, h], @"\\servidor\recurso\rec.cer", @"NT SERVICE\GPOS.Api");
        var (leido, motivo) = ContenidoArchivoAnillo.Leer(c.Serializar(), "GPOS.Api");
        Assert.Null(motivo);
        Assert.Equal(@"\\servidor\recurso\anillo", leido!.Carpeta);
    }

    /// <summary>
    /// SF-07: el descriptor de la llave sin <c>SE_DACL_PROTECTED</c> (heredable) pasa la comprobación del paso 9: <c>Comprobar</c> no mira el bit de
    /// protección que <c>Crear</c> sí pone.
    /// </summary>
    [Fact]
    public void SF07_el_descriptor_sin_DACL_protegida_pasa_la_comprobacion()
    {
        var sid = new System.Security.Principal.SecurityIdentifier("S-1-5-80-1111111111-2222222222-3333333333-4444444444-555555555");
        var protegido = GPOS.Migracion.Certificados.DescriptorLlave.Crear(sid);
        var raw = new System.Security.AccessControl.RawSecurityDescriptor(protegido, 0);
        Assert.True(raw.ControlFlags.HasFlag(System.Security.AccessControl.ControlFlags.DiscretionaryAclProtected));
        raw.SetFlags(raw.ControlFlags & ~System.Security.AccessControl.ControlFlags.DiscretionaryAclProtected);
        var sinProteger = new byte[raw.BinaryLength];
        raw.GetBinaryForm(sinProteger, 0);
        Assert.Null(GPOS.Migracion.Certificados.DescriptorLlave.Comprobar(sinProteger, sid));
    }
}
